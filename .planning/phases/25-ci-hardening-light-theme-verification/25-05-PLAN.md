The target — src/dashboard/index.html, the FIRST <script> tag (no `src` attribute), verbatim:
  <script>
    (function () {
      var STORAGE_KEY = 'dashboard-theme';
      var mode = 'auto';
      try {
        var raw = localStorage.getItem(STORAGE_KEY);
        if (raw === 'light' || raw === 'dark' || raw === 'auto') {
          mode = raw;
        }
      } catch (e) {
        mode = 'auto';
      }
      var effective = mode;
      if (mode === 'auto') {
        effective = window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light';
      }
      document.documentElement.setAttribute('data-theme', effective);
    })();

    `./theme.js` — assert the extracted script text contains that imported literal rather than
    a hard-coded `'dashboard-theme'` in the test, so a rename of the key in `theme.ts` breaks

  - the media query the bootstrap passes to `matchMedia` is exactly
    `'(prefers-color-scheme: dark)'`
