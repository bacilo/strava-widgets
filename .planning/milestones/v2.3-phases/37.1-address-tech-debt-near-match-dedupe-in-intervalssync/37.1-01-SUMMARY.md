```js
// SC5 probe: read-only. Reads reconciliation near rows and archive records, prints figures.
const fs = require('fs');
const path = require('path');
const root = process.cwd();
const g = JSON.parse(fs.readFileSync(path.join(root, 'data/reconciliation/garmin.json'), 'utf8'));
const actDir = path.join(root, 'data/activities');
const mismatches = [];
const near = g.near.map(r => ({ ...r }));
console.log(`near.length = ${near.length}`);
console.log(`counts.near = ${g.counts.near}`);
if (near.length !== g.counts.near) mismatches.push(`near.length ${near.length} !== counts.near ${g.counts.near}`);

if (process.env.SC5_PLANT_DELTA === '1') {
  if (near.length === 0) {
    console.log('PLANT VACUOUS: no near rows');
    process.exit(0);
  }
  near[0].delta_s = Number(near[0].delta_s) + 1;
}

for (const row of near) {
  let recomputed = NaN;
  let provider = 'absent';
  try {
    const a = JSON.parse(fs.readFileSync(path.join(actDir, `${row.archive_id}.json`), 'utf8'));
    recomputed = (Date.parse(row.start_date) - Date.parse(a.start_date)) / 1000;
    provider = a.source_provider === undefined ? 'absent' : a.source_provider;
  } catch (e) {
    mismatches.push(`row ${row.source_id}: archive ${row.archive_id} unreadable`);
  }
  const agree = recomputed === Number(row.delta_s);
  if (!agree) mismatches.push(`row ${row.source_id}: file delta ${row.delta_s} vs recomputed ${recomputed}`);
  const fit = provider === 'garmin-export';
  console.log(
    `row source_id=${row.source_id} archive_id=${row.archive_id} file_delta=${row.delta_s} ` +
      `recomputed=${recomputed} agree=${agree} source_provider=${provider} fit_imported=${fit} ` +
      `disposition=${fit ? 'sync would skip a delivery near it' : 'no change; record stays as is'}`
  );
}

const recs = [];
for (const f of fs.readdirSync(actDir).filter(x => x.endsWith('.json'))) {
  try {
    const a = JSON.parse(fs.readFileSync(path.join(actDir, f), 'utf8'));
    recs.push({ id: f.replace(/\.json$/, ''), epoch: Math.floor(Date.parse(a.start_date) / 1000), provider: a.source_provider });
  } catch (e) {
    /* unreadable: ignored in recount */
  }
}
const fits = recs.filter(r => r.provider === 'garmin-export');
console.log(`archive records total = ${recs.length}`);
console.log(`garmin-export records = ${fits.length} ids=[${fits.map(r => r.id).join(',')}]`);
let pairs = 0;
for (const f of fits) {
  for (const r of recs) {
    if (r.id === f.id || !Number.isFinite(r.epoch)) continue;
    const d = Math.abs(r.epoch - f.epoch);
    if (d >= 1 && d <= 120) {
      pairs++;
      console.log(`existing near pair: ${r.id} is ${r.epoch - f.epoch}s from FIT ${f.id}`);
    }
  }
}
console.log(`archive records within 1..120 s of a garmin-export record = ${pairs}`);

if (mismatches.length) {
  for (const m of mismatches) console.log(`SC5 PROBE MISMATCH: ${m}`);
  process.exitCode = 1;
} else {
  console.log('SC5 PROBE OK');
}
```
